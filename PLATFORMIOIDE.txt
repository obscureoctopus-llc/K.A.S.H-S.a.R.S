/**
 * K.A.S.H.S.A.R.S. Dual-Mode Tactical UI Panel
 * ============================================
 * Mode 0 (24GHz SAR Sonar): Renders active sweeping arcs with blip trail memory.
 * Mode 1 (60GHz Core Hex): Displays streaming raw hexadecimal blocks from the Seeed array.
 */

#include <Arduino.h>
#include <SPI.h>
#include <TFT_eSPI.h>

TFT_eSPI tft = TFT_eSPI();

// Operating Modes Selector 
enum OperationalMode { MODE_24G_SONAR, MODE_60G_DEBUG };
OperationalMode systemMode = MODE_24G_SONAR;

// Interface Sweep Configuration Settings
unsigned long lastSweepUpdate = 0;
int sweepAngle = 20;
int sweepDirection = 1;

// Target Blip History Arrays to provide smooth trailing persistence effects
#define MAX_BLIPS 30
struct RadarBlip {
    int x;
    int y;
    uint8_t intensity; // Higher value = newer blip
};
RadarBlip blipHistory[MAX_BLIPS];

// 24GHz Sensor Registers Payload Cache
bool targetPresent = false;
uint16_t movingDistanceCm = 0;
uint8_t movingTargetEnergy = 0;
uint8_t staticTargetEnergy = 0;
uint16_t primaryDistanceCm = 0;

unsigned long lastTouchDebounce = 0;
bool modeChanged = true;

// Forward Declarations
void handleTouchDetection();
void drawModeSelectionButton();
void run24G_SonarDisplay();
void run60G_HexDebugger();
void parse24GFrame(String dataLine);
void drawSonarStaticGrid();

void setup() {
  Serial.begin(115200);
  
  // Connect data line wire to CYD hardware serial 2 pins (RX=16, TX=17)
  Serial2.begin(115200, SERIAL_8N1, 16, 17); 

  tft.init();
  tft.setRotation(1); // Force default horizontal panoramic layout 
  tft.fillScreen(TFT_BLACK);

  // Initialize Blip History Struct Array Values
  for(int i = 0; i < MAX_BLIPS; i++) {
    blipHistory[i].intensity = 0;
  }
  
  // Calibrate internal resistive panel parameters
  uint16_t touchCalibrationData[5] = { 287, 3450, 483, 3314, 1 };
  tft.setTouch(touchCalibrationData);
  
  Serial.println(F("[KASH_SARS] Tactical Dual-Mode Control OS Engine Mounted."));
}

void loop() {
  handleTouchDetection();

  if (systemMode == MODE_24G_SONAR) {
    if (modeChanged) {
      tft.fillScreen(TFT_BLACK);
      drawSonarStaticGrid();
      modeChanged = false;
    }
    
    // Snag telemetry strings incoming on the wire
    if (Serial2.available() > 0) {
      String frameString = Serial2.readStringUntil('\n');
      if (frameString.startsWith("R,")) {
        parse24GFrame(frameString);
      }
    }
    
    // Cycle sweep line arrays independent of parsing speeds
    if (millis() - lastSweepUpdate >= 20) {
      lastSweepUpdate = millis();
      run24G_SonarDisplay();
    }
  } 
  else if (systemMode == MODE_60G_DEBUG) {
    if (modeChanged) {
      tft.fillScreen(TFT_BLACK);
      tft.setTextColor(TFT_YELLOW, TFT_BLACK);
      tft.drawString("60GHz CORE HEX CONSOLE DEBUGGER", 10, 45, 2);
      tft.drawFastHLine(10, 62, 300, TFT_YELLOW);
      modeChanged = false;
    }
    run60G_HexDebugger();
  }
}

void handleTouchDetection() {
  uint16_t touchX = 0, touchY = 0;
  
  // Query resistive glass status registers
  if (tft.getTouch(&touchX, &touchY)) {
    if (millis() - lastTouchDebounce > 350) { // Software Debounce Frame Block
      lastTouchDebounce = millis();
      
      // Target area: Top Right Toggle Box (X: 200-320, Y: 0-40)
      if (touchX > 200 && touchY < 40) {
        if (systemMode == MODE_24G_SONAR) {
          systemMode = MODE_60G_DEBUG;
        } else {
          systemMode = MODE_24G_SONAR;
        }
        modeChanged = true;
      }
    }
  }
  
  // Draw the button persistently on top of all active layers
  drawModeSelectionButton();
}

void drawModeSelectionButton() {
  tft.drawRect(210, 5, 105, 30, TFT_WHITE);
  tft.fillRect(211, 6, 103, 28, TFT_DARKGREY);
  tft.setTextColor(TFT_WHITE);
  tft.setTextDatum(MC_DATUM);
  
  if (systemMode == MODE_24G_SONAR) {
    tft.drawString("GO TO 60G", 262, 20, 2);
  } else {
    tft.drawString("GO TO 24G", 262, 20, 2);
  }
  tft.setTextDatum(TL_DATUM); // Restore standard text reference points
}

void parse24GFrame(String dataLine) {
  int comma1 = dataLine.indexOf(',');
  int comma2 = dataLine.indexOf(',', comma1 + 1);
  int comma3 = dataLine.indexOf(',', comma2 + 1);
  int comma4 = dataLine.indexOf(',', comma3 + 1);
  int comma5 = dataLine.indexOf(',', comma4 + 1);
  int comma6 = dataLine.indexOf(',', comma5 + 1);

  if (comma6 != -1) {
    targetPresent       = dataLine.substring(comma1 + 1, comma2).toInt();
    movingDistanceCm   = dataLine.substring(comma2 + 1, comma3).toInt();
    movingTargetEnergy   = dataLine.substring(comma3 + 1, comma4).toInt();
    staticTargetEnergy   = dataLine.substring(comma4 + 1, comma5).toInt();
    primaryDistanceCm   = dataLine.substring(comma6 + 1).toInt();
  }
}

void drawSonarStaticGrid() {
  tft.setTextColor(TFT_GREEN, TFT_BLACK);
  tft.drawString("K.A.S.H.S.A.R.S. SONAR V2", 10, 10, 2);
  
  // Coordinates anchored around bottom horizontal midplane (160, 240)
  tft.drawCircle(160, 240, 200, TFT_DARKGREY);
  tft.drawCircle(160, 240, 140, TFT_DARKGREY);
  tft.drawCircle(160, 240, 80, TFT_DARKGREY);
}

void run24G_SonarDisplay() {
  int oX = 160, oY = 240, r = 210;

  // Erase behind current sweeping array track
  float clearRad = (sweepAngle - (2 * sweepDirection)) * PI / 180.0;
  tft.drawLine(oX, oY, oX + r * cos(clearRad), oY - r * sin(clearRad), TFT_BLACK);

  // Sweep line increments
  sweepAngle += sweepDirection;
  if (sweepAngle >= 160 || sweepAngle <= 20) sweepDirection = -sweepDirection;

  float currentRad = sweepAngle * PI / 180.0;
  int beamEndX = oX + r * cos(currentRad);
  int beamEndY = oY - r * sin(currentRad);

  // Redraw structural tracking grid sectors inside active field lines
  tft.drawCircle(oX, oY, 200, TFT_DARKGREY);
  tft.drawCircle(oX, oY, 140, TFT_DARKGREY);
  tft.drawCircle(oX, oY, 80, TFT_DARKGREY);
  
  // Draw primary scanner vector tracking line
  tft.drawLine(oX, oY, beamEndX, beamEndY, TFT_DARKGREEN);

  // Manage target plotting if response frame breaks filtering limits
  if (targetPresent && primaryDistanceCm > 15) {
    float scale = (float)primaryDistanceCm / 500.0; // 500 cm tracking scaling threshold
    if (scale > 1.0) scale = 1.0;

    int bRadius = r * scale;
    int blipX = oX + bRadius * cos(currentRad);
    int blipY = oY - bRadius * sin(currentRad);

    // Append newly captured trace directly to tracking matrix arrays
    for (int i = 0; i < MAX_BLIPS; i++) {
      if (blipHistory[i].intensity == 0) {
        blipHistory[i].x = blipX;
        blipHistory[i].y = blipY;
        blipHistory[i].intensity = 255; // Maximum persistence signature
        break;
      }
    }
  }

  // Blip History Engine Loop: Draw trails and decay their brightness over time
  for (int i = 0; i < MAX_BLIPS; i++) {
    if (blipHistory[i].intensity > 0) {
      uint16_t blipColor;
      
      // Isolate color states to evaluate victim hazard conditions
      if (movingTargetEnergy == 0 && staticTargetEnergy > 10) {
        blipColor = tft.color565(blipHistory[i].intensity, 0, 0); // Decaying Red (Trapped)
      } else {
        blipColor = tft.color565(0, blipHistory[i].intensity, 0); // Decaying Green (Moving)
      }

      tft.fillCircle(blipHistory[i].x, blipHistory[i].y, 3, blipColor);
      
      // Decay multiplier factor per loop cycle iteration
      if(blipHistory[i].intensity > 15) blipHistory[i].intensity -= 15;
      else {
        // Clear artifact footprint cleanly once threshold signature falls off
        tft.fillCircle(blipHistory[i].x, blipHistory[i].y, 3, TFT_BLACK);
        blipHistory[i].intensity = 0;
      }
    }
  }

  // Update text measurements matrix arrays cleanly inside top-left box
  char metricsText;
  snprintf(metricsText, sizeof(metricsText), "%3ucm  %.1fm ", primaryDistanceCm, (float)primaryDistanceCm / 100.0);
  tft.setTextColor(TFT_GREEN, TFT_BLACK);
  tft.drawString(metricsText, 10, 30, 4);
}

void run60G_HexDebugger() {
  static int consoleRow = 70;
  
  // Capture raw data characters coming directly from the Seeed Studio board UART loop
  while (Serial2.available() > 0) {
    uint8_t hexByte = Serial2.read();

    tft.setTextColor(TFT_GREEN, TFT_BLACK);
    char debugChar;
    snprintf(debugChar, sizeof(debugChar), "%02X ", hexByte);
    
    // Step cursor down screen canvas manually to form data block matrices
    static int cursorX = 10;
    tft.drawString(debugChar, cursorX, consoleRow, 2);
    cursorX += 26;

    if (cursorX > 300) {
      cursorX = 10;
      consoleRow += 16;
      
      // Clear console if it runs off the bottom margin of the screen frame
      if (consoleRow > 220) {
        tft.fillRect(10, 70, 300, 160, TFT_BLACK);
        consoleRow = 70;
      }
    }
  }
}
